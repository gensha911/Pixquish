    25→    25→/** Encodes a canvas to a blob, wrapped as a promise. */
    26→    26→function toBlob(
    27→    27→  canvas: HTMLCanvasElement,
    28→    28→  format: string,
    29→    29→  quality: number,
    30→    30→): Promise<Blob | null> {
    31→    31→  return new Promise((resolve) => {
    32→    32→    canvas.toBlob(
    33→    33→      (blob) => resolve(blob),
    34→    34→      format,
    35→    35→      Math.min(1, Math.max(0, quality)),
    36→    36→    );
    37→    37→  });
    38→    38→}

    40→    40→/** Whether the format is natively lossless (quality param has no effect on output). */
    41→    41→function isLosslessFormat(format: string): boolean {
    42→    42→  return format === "image/png";
    43→    43→}
    44→    44→
    45→    45→/** Whether the format is a slow encoder (AVIF is ~5-10x slower than JPEG/WebP). */
    46→    46→function isSlowFormat(format: string): boolean {
    47→    47→  return format === "image/avif";
    48→    48→}

    50→    50→/** Draws the source into a canvas at the given scale and returns it.
    51→    51→ *  For lossless formats (PNG), image smoothing is disabled to preserve
    52→    52→ *  pixel-perfect sharpness. */
    53→    53→function renderCanvas(
    54→    54→  source: CanvasImageSource,
    55→    55→  origW: number,
    56→    56→  origH: number,
    57→    57→  scale: number,
    58→    58→  format: string,
    59→    59→): { canvas: HTMLCanvasElement; w: number; h: number } {
    60→    60→  const w = Math.max(1, Math.round(origW * scale));
    61→    61→  const h = Math.max(1, Math.round(origH * scale));
    62→    62→  const canvas = document.createElement("canvas");
    63→    63→  canvas.width = w;
    64→    64→  canvas.height = h;
    65→    65→  const ctx = canvas.getContext("2d");
    66→    66→  if (!ctx) throw new Error("Canvas 2D context unavailable.");
    67→    67→  // Lossless formats: disable smoothing for pixel-perfect output.
    68→    68→  // Lossy formats (JPEG, WebP, AVIF): enable high-quality smoothing.
    69→    69→  const isLossless = isLosslessFormat(format);
    70→    70→  const smoothing = !(isLossless && scale >= 1);
    71→    71→
    72→    72→  // Multi-step downscale: much sharper than a single large scale step.
    73→    73→  // Reduces in 50% increments, each preserving more detail.
    74→    74→  if (smoothing && scale < 0.85) {
    75→    75→    let currentSource: CanvasImageSource = source;
    76→    76→    let curW = origW;
    77→    77→    let curH = origH;
    78→    78→    const canvasEl = document.createElement('canvas');
    79→    79→    canvasEl.width = w;
    80→    80→    canvasEl.height = h;
    81→    81→    const ctx = canvasEl.getContext('2d')!;
    82→    82→    ctx.imageSmoothingEnabled = true;
    83→    83→    ctx.imageSmoothingQuality = 'high';
    84→    84→    // We draw in steps — the final draw goes directly to the output canvas
    85→    85→    while (curW / 2 > w && curH / 2 > h) {
    86→    86→      const nextW = Math.max(w, Math.round(curW / 2));
    87→    87→      const nextH = Math.max(h, Math.round(curH / 2));
    88→    88→      const tempCanvas = document.createElement('canvas');
    89→    89→      tempCanvas.width = nextW;
    90→    90→      tempCanvas.height = nextH;
    91→    91→      const tempCtx = tempCanvas.getContext('2d')!;
    92→    92→      tempCtx.imageSmoothingEnabled = true;
    93→    93→      tempCtx.imageSmoothingQuality = 'high';
    94→    94→      tempCtx.drawImage(currentSource, 0, 0, nextW, nextH);
    95→    95→      currentSource = tempCanvas;
    96→    96→      curW = nextW;
    97→    97→      curH = nextH;
    98→    98→    }
    99→    99→    ctx.drawImage(currentSource, 0, 0, w, h);
   100→   100→    return { canvas: canvasEl, w, h };
   101→   101→  }
   102→   102→
   103→   103→  if (isLossless && scale >= 1) {
   104→   104→    ctx.imageSmoothingEnabled = false;
   105→   105→  } else {
   106→   106→    ctx.imageSmoothingEnabled = true;
   107→   107→    ctx.imageSmoothingQuality = 'high';
   108→   108→  }
   109→   109→  ctx.drawImage(source, 0, 0, w, h);
   110→   110→  return { canvas, w, h };
   111→   111→}
   112→   112→
   113→   113→async function encode(
   114→   114→  source: CanvasImageSource,
   115→   115→  origW: number,
   116→   116→  origH: number,
   117→   117→  scale: number,
   118→   118→  format: string,
   119→   119→  quality: number,
   120→   120→): Promise<EncodeCandidate | null> {
   121→   121→  const { canvas, w, h } = renderCanvas(source, origW, origH, scale, format);
   122→   122→  // For JPEG, fill alpha with white so transparency doesn't turn black.
   123→   123→  if (format === "image/jpeg") {
   124→   124→    const ctx = canvas.getContext("2d");
   125→   125→    if (ctx) {
   126→   126→      ctx.globalCompositeOperation = "destination-over";
   127→   127→      ctx.fillStyle = "#ffffff";
   128→   128→      ctx.fillRect(0, 0, w, h);
   129→   129→      ctx.globalCompositeOperation = "source-over";
   130→   130→    }
   131→   131→  }
   132→   132→  const blob = await toBlob(canvas, format, quality);
   133→   133→  if (!blob) return null;
   134→   134→  return { blob, quality, scale, width: w, height: h };
   135→   135→}
   136→   136→
   137→   137→/** Quick feature-detection that the browser can encode a given format. */
   138→   138→let formatSupportCache: Record<string, boolean> | null = null;
   139→   139→async function supportsFormat(format: string): Promise<boolean> {
   140→   140→  if (format === "image/png") return true; // universally supported
   141→   141→  if (formatSupportCache) return formatSupportCache[format] ?? false;
   142→   142→  const cache: Record<string, boolean> = {};
   143→   143→  const probe = document.createElement("canvas");
   144→   144→  probe.width = 1;
   145→   145→  probe.height = 1;
   146→   146→  for (const f of ["image/jpeg", "image/webp", "image/avif"]) {
   147→   147→    const blob = await new Promise<Blob | null>((res) =>
   148→   148→      probe.toBlob((b) => res(b), f, 0.8),
   149→   149→    );
   150→   150→    cache[f] = !!blob && blob.size > 0;
   151→   151→  }
   152→   152→  formatSupportCache = cache;
   153→   153→  return cache[format] ?? false;
   154→   154→}

   156→   156→/** Compresses a single image file according to the provided options.
   157→   157→ *  If `cachedSource` is provided, skips image decoding (big speedup on re-compress).
   158→   158→ *  If `cachedAnalysis` is provided, skips pixel analysis.
   159→   159→ *  If `compressOpts.formatVerified` is true, skips the async format-support probe.
   160→   160→ *  In "max" mode with a target size, the output is guaranteed to not exceed the target. */
   161→   161→export async function compressImage(
   162→   162→  file: File,
   163→   163→  options: CompressionOptions,
   164→   164→  onProgress?: (progress: number) => void,
   165→   165→  cachedSource?: { source: CanvasImageSource & { width?: number; height?: number }; width: number; height: number } | null,
   166→   166→  cachedAnalysis?: ImageAnalysis | null,
   167→   167→  compressOpts?: { formatVerified?: boolean },
   168→   168→): Promise<CompressionResult> {
   169→   169→  const start = performance.now();
   170→   170→  const alreadyVerified = compressOpts?.formatVerified === true;
   171→   171→
   172→   172→  let source: CanvasImageSource & { width?: number; height?: number };
   173→   173→  let width: number;
   174→   174→  let height: number;
   175→   175→  let analysis: ImageAnalysis;
   176→   176→
   177→   177→  if (cachedSource) {
   178→   178→    source = cachedSource.source;
   179→   179→    width = cachedSource.width;
   180→   180→    height = cachedSource.height;
   181→   181→  } else {
   182→   182→    const loaded = await loadImageSource(file);
   183→   183→    source = loaded.source;
   184→   184→    width = loaded.width;
   185→   185→    height = loaded.height;
   186→   186→  }
   187→   187→  onProgress?.(0.1);
   188→   188→
   189→   189→  if (cachedAnalysis) {
   190→   190→    analysis = cachedAnalysis;
   191→   191→  } else {
   192→   192→    analysis = analyzeImage(file, source, width, height);
   193→   193→  }
   194→   194→  onProgress?.(0.2);
   195→   195→
   196→   196→  let settings = recommendSettings(analysis, options);
   197→   197→
   198→   198→  // Ensure the chosen format is encodable; otherwise fall back gracefully.
   199→   199→  // Skip the async probe if we already verified on a previous compress.
   200→   200→  if (!alreadyVerified && !(await supportsFormat(settings.format))) {
   201→   201→    settings = {
   202→   202→      ...settings,
   203→   203→      format: analysis.hasTransparency ? "image/png" : "image/jpeg",
   204→   204→      reason:
   205→   205→        settings.reason +
   206→   206→        ` (Requested format unsupported in this browser — fell back to ${
   207→   207→          analysis.hasTransparency ? "PNG" : "JPG"
   208→   208→        }.)`,
   209→   209→    };
   210→   210→  }
   211→   211→  onProgress?.(0.25);
   212→   212→
   213→   213→  const origW = width;
   214→   214→  const origH = height;
   215→   215→  let note: string | undefined;
   216→   216→
   217→   217→  // In max mode the user's chosen target size is a hard cap.
   218→   218→  const enforceTarget = options.mode === "max" && !!options.targetSize && options.targetSize > 0;
   219→   219→
   220→   220→  // --- Target-size mode: binary search over quality (and scale if needed) ---
   221→   221→  let candidate: EncodeCandidate | null = null;
   222→   222→  let targetMet: boolean | null = null;
   223→   223→
   224→   224→  if (options.targetSize && options.targetSize > 0 && options.targetSize < file.size) {
   225→   225→    const target = options.targetSize;
   226→   226→    const floor = qualityFloor(options.mode);
   227→   227→    // For lossless formats (PNG) the only way to shrink is downscaling,
   228→   228→    // so allow much more aggressive resize than the mode's usual floor.
   229→   229→    const isLossless = isLosslessFormat(settings.format);
   230→   230→    const sFloor = isLossless ? 0.1 : scaleFloor(options.mode);
   231→   231→
   232→   232→    const search = await searchTargetSize(
   233→   233→      source,
   234→   234→      origW,
   235→   235→      origH,
   236→   236→      settings.format,
   237→   237→      target,
   238→   238→      floor,
   239→   239→      sFloor,
   240→   240→      (p) => onProgress?.(0.25 + p * 0.65),
   241→   241→      isLossless, // skip quality search for lossless formats
   242→   242→      enforceTarget,
   243→   243→    );
   244→   244→
   245→   245→    candidate = search.best;
   246→   246→    targetMet = search.met;
   247→   247→    if (!search.met) {
   248→   248→      note = search.note;
   249→   249→    }
   250→   250→    settings = {
   251→   251→      ...settings,
   252→   252→      quality: candidate ? candidate.quality : settings.quality,
   253→   253→      resizeScale: candidate ? candidate.scale : settings.resizeScale,
   254→   254→    };
   255→   255→  } else {
   256→   256→    // --- Standard mode: single encode at recommended quality ---
   257→   257→    // If target is set but >= original, the file is already under target.
   258→   258→    if (options.targetSize && options.targetSize >= file.size) {
   259→   259→      targetMet = true;
   260→   260→    }
   261→   261→    candidate = await encode(
   262→   262→      source,
   263→   263→      origW,
   264→   264→      origH,
   265→   265→      settings.resizeScale,
   266→   266→      settings.format,
   267→   267→      settings.quality,
   268→   268→    );
   269→   269→    onProgress?.(0.7);
   270→   270→  }
   271→   271→
   272→   272→  // When auto format + lossless couldn't reach target, fall back to lossy WebP.
   273→   273→  // But NEVER override an explicit user format choice.
   274→   274→  if (
   275→   275→    options.targetSize &&
   276→   276→    options.format === "auto" &&
   277→   277→    isLosslessFormat(settings.format) &&
   278→   278→    candidate &&
   279→   279→    candidate.blob.size > options.targetSize &&
   280→   280→    (await supportsFormat("image/webp"))
   281→   281→  ) {
   282→   282→    const target = options.targetSize;
   283→   283→    const floor = qualityFloor(options.mode);
   284→   284→    const sFloor = scaleFloor(options.mode);
   285→   285→    const webpSearch = await searchTargetSize(
   286→   286→      source,
   287→   287→      origW,
   288→   288→      origH,
   289→   289→      "image/webp",
   290→   290→      target,
   291→   291→      floor,
   292→   292→      sFloor,
   293→   293→      (p) => onProgress?.(0.25 + p * 0.65),
   294→   294→      false,
   295→   295→      enforceTarget,
   296→   296→    );
   297→   297→    if (webpSearch.best && webpSearch.best.blob.size < candidate.blob.size) {
   298→   298→      candidate = webpSearch.best;
   299→   299→      targetMet = webpSearch.met;
   300→   300→      settings = {
   301→   301→        ...settings,
   302→   302→        format: "image/webp",
   303→   303→        quality: candidate.quality,
   304→   304→        resizeScale: candidate.scale,
   305→   305→        reason:
   306→   306→          settings.reason +
   307→   307→          " Switched to WebP lossy to reach the target size (lossless format stayed too large).",
   308→   308→      };
   309→   309→      if (!webpSearch.met) note = webpSearch.note;
   310→   310→    }
   311→   311→  }
   312→   312→
   313→   313→  // Release the image source only if we created it (not cached).
   314→   314→  if (!cachedSource && "close" in source && typeof (source as ImageBitmap).close === "function") {
   315→   315→    (source as ImageBitmap).close();
   316→   316→  }
   317→   317→
   318→   318→  if (!candidate) {
   319→   319→    throw new Error("Compression failed: the browser could not encode this image.");
   320→   320→  }
   321→   321→
   322→   322→  onProgress?.(0.85);
   323→   323→
   324→   324→  const originalSize = file.size;
   325→   325→  const originalUrl = URL.createObjectURL(file);
   326→   326→
   327→   327→  // If the compressed output is not smaller than the original, keep the
   328→   328→  // original file (it's already well optimized) and explain honestly.
   329→   329→  // EXCEPTION: in max mode with a target size, we must respect the target
   330→   330→  // even if the original is smaller than the compressed output.
   331→   331→  let finalBlob: Blob = candidate.blob;
   332→   332→  let finalWidth = candidate.width;
   333→   333→  let finalHeight = candidate.height;
   334→   334→  let finalFormat = settings.format;
   335→   335→  let finalQuality = candidate.quality;
   336→   336→  let keptOriginal = false;
   337→   337→
   338→   338→  // Whether the user explicitly chose a different output format.
   339→   339→  // When true, we always honour the new format even if the result is larger.
   340→   340→  const formatChanged =
   341→   341→    options.format !== "original" &&
   342→   342→    options.format !== "auto" &&
   343→   343→    settings.format !== file.type;
   344→   344→
   345→   345→  // Never let output exceed the original file size when the format hasn't changed.
   346→   346→  // The enforceTarget bypass only applies when the target is actually smaller
   347→   347→  // than the original (meaning the user genuinely wants to shrink the file).
   348→   348→  const targetIsSmaller = (options.targetSize ?? 0) > 0 && (options.targetSize ?? 0) < originalSize;
   349→   349→  if (candidate.blob.size >= originalSize && (!enforceTarget || !targetIsSmaller) && !formatChanged) {
   350→   350→    finalBlob = file;
   351→   351→    finalWidth = origW;
   352→   352→    finalHeight = origH;
   353→   353→    finalFormat = file.type || settings.format;
   354→   354→    finalQuality = 1;
   355→   355→    keptOriginal = true;
   356→   356→    note =
   357→   357→      "This image is already well optimized — keeping the original to avoid making it larger.";
   358→   358→  }
   359→   359→
   360→   360→  // When format was explicitly changed, always output the new format even if larger.
   361→   361→  // Add an informative note so the user understands why the file grew.
   362→   362→  if (formatChanged && candidate.blob.size >= originalSize && !note) {
   363→   363→    const diff = formatBytes(candidate.blob.size - originalSize);
   364→   364→    note = `Converted to ${shortFormat(settings.format)} as requested. File is ${diff} larger (format differences).`;
   365→   365→  }
   366→   366→
   367→   367→  const url = URL.createObjectURL(finalBlob);
   368→   368→  const size = finalBlob.size;
   369→   369→  const savedBytes = Math.max(0, originalSize - size);
   370→   370→  const savedPercent =
   371→   371→    originalSize > 0 ? (savedBytes / originalSize) * 100 : 0;
   372→   372→  const loadImprovement =
   373→   373→    size > 0 ? originalSize / size : 1;
   374→   374→
   375→   375→  onProgress?.(1);
   376→   376→
   377→   377→  return {
   378→   378→    blob: finalBlob,
   379→   379→    url,
   380→   380→    width: finalWidth,
   381→   381→    height: finalHeight,
   382→   382→    format: finalFormat,
   383→   383→    quality: finalQuality,
   384→   384→    size,
   385→   385→    settings: {
   386→   386→      ...settings,
   387→   387→      quality: finalQuality,
   388→   388→      reason: keptOriginal
   389→   389→        ? "Original kept (already optimal)."
   390→   390→        : settings.reason,
   391→   391→    },
   392→   392→    analysis,
   393→   393→    originalSize,
   394→   394→    originalUrl,
   395→   395→    savedBytes,
   396→   396→    savedPercent,
   397→   397→    loadImprovement,
   398→   398→    targetMet,
   399→   399→    targetSize: options.targetSize ?? null,
   400→   400→    note,
   401→   401→    durationMs: Math.round(performance.now() - start),
   402→   402→  };
   403→   403→}
   404→   404→
   405→   405→interface SearchOutcome {
   406→   406→  best: EncodeCandidate | null;
   407→   407→  met: boolean;
   408→   408→  note?: string;
   409→   409→}
   410→   410→
   411→   411→/** Tolerance band: results within this fraction of the target (from below)
   412→   412→ *  are considered "ideally close" and preferred over any result that is
   413→   413→ *  significantly smaller (which would indicate over-compression).
   414→   414→ *  0.97 means a result >= 97% of target is considered near-ideal. */
   415→   415→const TARGET_PROXIMITY = 0.97;
   416→   416→
   417→   417→/** Maximum quality ceiling used during searches. Keeping this just under 1.0
   418→   418→ *  avoids browser encoders that round quality up to lossless at 1.0. */
   419→   419→const QUALITY_CEILING = 0.95;
   420→   420→
   421→   421→/** Quality floor for the aggressive (Phase 3) search in Max mode.
   422→   422→ *  Kept above 0.01 so we never produce a completely degenerate image. */
   423→   423→const AGGRESSIVE_QUALITY_FLOOR = 0.05;
   424→   424→
   425→   425→/** Improved binary search over quality (and progressively over scale) to
   426→   426→ *  approach a target byte size as closely as possible WITHOUT unnecessary
   427→   427→ *  over-compression.
   428→   428→ *
   429→   429→ *  Design goals (Max Compress + target size):
   430→   430→ *  1. Converge as close as possible to the target from below — never drop
   431→   431→ *     well below the target if a higher-quality setting still fits.
   432→   432→ *  2. Use fine-grained quality steps so visual quality doesn't collapse.
   433→   433→ *  3. Prefer raising quality at the current scale before shrinking the image.
   434→   434→ *
   435→   435→ *  When `enforceTarget` is true (Max Compress mode), the function guarantees
   436→   436→ *  the returned blob will be at or under the target — even if that requires
   437→   437→ *  quality reduction and downscaling. The search still maximises quality
   438→   438→ *  within that constraint. */
   439→   439→async function searchTargetSize(
   440→   440→  source: CanvasImageSource,
   441→   441→  origW: number,
   442→   442→  origH: number,
   443→   443→  format: string,
   444→   444→  target: number,
   445→   445→  floor: number,
   446→   446→  sFloor: number,
   447→   447→  onProgress?: (p: number) => void,
   448→   448→  skipQualitySearch = false,
   449→   449→  enforceTarget = false,
   450→   450→): Promise<SearchOutcome> {
   451→   451→  let bestUnder: EncodeCandidate | null = null;
   452→   452→  let bestAbove: EncodeCandidate | null = null;
   453→   453→  const report = (frac: number) => onProgress?.(frac);
   454→   454→
   455→   455→  // Track whether we already have a near-ideal result (within the proximity
   456→   456→  // band). Once we do, subsequent phases can stop early to avoid wasting
   457→   457→  // encodes and to avoid replacing a near-ideal result with an over-compressed
   458→   458→  // one from a smaller scale.
   459→   459→  const isNearIdeal = (c: EncodeCandidate | null): boolean =>
   460→   460→    !!c && c.blob.size >= target * TARGET_PROXIMITY && c.blob.size <= target;
   461→   461→
   462→   462→  // ─────────────────────────────────────────────────────────────────────
   463→   463→  // Phase 1: quality binary search at FULL resolution (scale = 1).
   464→   464→  // This is the highest-fidelity path — we only downscale if quality alone
   465→   465→  // cannot reach the target.
   466→   466→  // ─────────────────────────────────────────────────────────────────────
   467→   467→  if (!skipQualitySearch) {
   468→   468→    const slow = isSlowFormat(format);
   469→   469→    // Finer granularity than before: 12 steps for fast formats gives
   470→   470→    // ~0.0007 quality precision (vs 0.0019 at 9 steps). AVIF stays lean.
   471→   471→    let lo = floor;
   472→   472→    let hi = QUALITY_CEILING;
   473→   473→    const steps = slow ? 5 : 12;
   474→   474→
   475→   475→    // For AVIF: quick 3-point probe to narrow the range before binary search.
   476→   476→    if (slow) {
   477→   477→      const probeQualities = [0.3, 0.6, 0.9];
   478→   478→      const probeCandidates: EncodeCandidate[] = [];
   479→   479→      for (const pq of probeQualities) {
   480→   480→        const pc = await encode(source, origW, origH, 1, format, pq);
   481→   481→        if (pc) probeCandidates.push(pc);
   482→   482→      }
   483→   483→      report(0.2);
   484→   484→
   485→   485→      probeCandidates.sort((a, b) => b.quality - a.quality);
   486→   486→      for (let pi = 0; pi < probeCandidates.length; pi++) {
   487→   487→        const pr = probeCandidates[pi];
   488→   488→        if (pr.blob.size <= target) {
   489→   489→          lo = pr.quality;
   490→   490→          if (pi > 0) hi = probeCandidates[pi - 1].quality;
   491→   491→          bestUnder = pickBetterUnder(bestUnder, pr, target);
   492→   492→          break;
   493→   493→        } else {
   494→   494→          bestAbove = pickSmallerAbove(bestAbove, pr);
   495→   495→        }
   496→   496→      }
   497→   497→      if (!bestUnder && probeCandidates.length > 0) {
   498→   498→        const minQ = Math.min(...probeCandidates.map(c => c.quality));
   499→   499→        hi = minQ;
   500→   500→      }
   501→   501→    }
   502→   502→
   503→   503→    for (let i = 0; i < steps; i++) {
   504→   504→      const mid = (lo + hi) / 2;
   505→   505→      const cand = await encode(source, origW, origH, 1, format, mid);
   506→   506→      report(Math.min(0.5, 0.2 + 0.25 * ((i + 1) / steps)));
   507→   507→      if (i % 3 === 2) await yieldToMain();
   508→   508→      if (!cand) continue;
   509→   509→
   510→   510→      if (cand.blob.size <= target) {
   511→   511→        bestUnder = pickBetterUnder(bestUnder, cand, target);
   512→   512→        lo = mid; // under target → push quality higher
   513→   513→        // Early exit: within the proximity band, we're essentially at target.
   514→   514→        if (isNearIdeal(bestUnder)) {
   515→   515→          // Refine a little more around this quality to squeeze out maximum
   516→   516→          // quality while staying under target, then return.
   517→   517→          bestUnder = await refineAroundBest(
   518→   518→            source, origW, origH, 1, format, bestUnder, lo, hi, target, report, 0.5, 0.6,
   519→   519→          );
   520→   520→          report(0.6);
   521→   521→          return { best: bestUnder, met: true };
   522→   522→        }
   523→   523→      } else {
   524→   524→        bestAbove = pickSmallerAbove(bestAbove, cand);
   525→   525→        hi = mid;
   526→   526→      }
   527→   527→    }
   528→   528→
   529→   529→    if (bestUnder) {
   530→   530→      // Phase 1 met the target (possibly well under). Refine upward to
   531→   531→      // recover any quality we can without exceeding the target.
   532→   532→      bestUnder = await refineAroundBest(
   533→   533→        source, origW, origH, 1, format, bestUnder, bestUnder.quality, QUALITY_CEILING, target, report, 0.55, 0.62,
   534→   534→      );
   535→   535→      report(0.62);
   536→   536→      return { best: bestUnder, met: true };
   537→   537→    }
   538→   538→  } else {
   539→   539→    // Lossless (PNG): quality has no effect, so a single full-res encode
   540→   540→    // establishes the baseline. Only downscaling can reduce size.
   541→   541→    const baseline = await encode(source, origW, origH, 1, format, 1);
   542→   542→    if (baseline) {
   543→   543→      if (baseline.blob.size <= target) {
   544→   544→        report(0.55);
   545→   545→        return { best: baseline, met: true };
   546→   546→      }
   547→   547→      bestAbove = baseline;
   548→   548→    }
   549→   549→    report(0.1);
   550→   550→  }
   551→   551→
   552→   552→  // ─────────────────────────────────────────────────────────────────────
   553→   553→  // Phase 2: progressive downscale with quality binary search per scale.
   554→   554→  // Only reached when quality alone at full resolution can't hit the target.
   555→   555→  //
   556→   556→  // Key improvement: each scale that finds a candidate under target is
   557→   557→  // refined upward in quality (not just accepted at the binary-search
   558→   558→  // midpoint). We also stop early if a near-ideal result is found, rather
   559→   559→  // than continuing to shrink unnecessarily.
   560→   560→  // ─────────────────────────────────────────────────────────────────────
   561→   561→  const slow = isSlowFormat(format);
   562→   562→  const allScales = skipQualitySearch
   563→   563→    ? [0.9, 0.85, 0.8, 0.75, 0.7, 0.65, 0.6, 0.55, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1]
   564→   564→    : slow
   565→   565→      ? [0.85, 0.75, 0.65, 0.55, 0.45]
   566→   566→      : [0.9, 0.85, 0.8, 0.75, 0.7, 0.65, 0.6];
   567→   567→  const scales = allScales.filter((s) => s >= sFloor);
   568→   568→
   569→   569→  for (let si = 0; si < scales.length; si++) {
   570→   570→    const scale = scales[si];
   571→   571→    const frac = 0.62 + 0.25 * ((si + 1) / scales.length);
   572→   572→
   573→   573→    if (skipQualitySearch) {
   574→   574→      const cand = await encode(source, origW, origH, scale, format, 1);
   575→   575→      report(frac);
   576→   576→      if (!cand) continue;
   577→   577→      if (cand.blob.size <= target) {
   578→   578→        bestUnder = pickBetterUnder(bestUnder, cand, target);
   579→   579→        // Near-ideal lossless result — stop shrinking.
   580→   580→        if (isNearIdeal(bestUnder)) {
   581→   581→          report(0.9);
   582→   582→          return { best: bestUnder, met: true };
   583→   583→        }
   584→   584→      } else {
   585→   585→        bestAbove = pickSmallerAbove(bestAbove, cand);
   586→   586→      }
   587→   587→    } else {
   588→   588→      // Binary search quality at this scale, then refine upward.
   589→   589→      let lo = floor;
   590→   590→      let hi = QUALITY_CEILING;
   591→   591→      const stepsPerScale = slow ? 4 : 8;
   592→   592→      for (let i = 0; i < stepsPerScale; i++) {
   593→   593→        const mid = (lo + hi) / 2;
   594→   594→        const cand = await encode(source, origW, origH, scale, format, mid);
   595→   595→        report(frac - 0.03 + 0.03 * ((i + 1) / stepsPerScale));
   596→   596→        if (!cand) continue;
   597→   597→        if (cand.blob.size <= target) {
   598→   598→          bestUnder = pickBetterUnder(bestUnder, cand, target);
   599→   599→          lo = mid;
   600→   600→          if (isNearIdeal(bestUnder)) break;
   601→   601→        } else {
   602→   602→          bestAbove = pickSmallerAbove(bestAbove, cand);
   603→   603→          hi = mid;
   604→   604→        }
   605→   605→        await yieldToMain();
   606→   606→      }
   607→   607→
   608→   608→      // Refine upward at this scale to maximise quality under target.
   609→   609→      if (bestUnder && bestUnder.scale === scale) {
   610→   610→        bestUnder = await refineAroundBest(
   611→   611→          source, origW, origH, scale, format, bestUnder, bestUnder.quality, QUALITY_CEILING, target, report, frac, frac + 0.02,
   612→   612→        );
   613→   613→      }
   614→   614→
   615→   615→      // If we have a near-ideal result, stop shrinking — further scales
   616→   616→      // would only reduce resolution without meaningful size gain.
   617→   617→      if (isNearIdeal(bestUnder)) {
   618→   618→        report(0.9);
   619→   619→        return { best: bestUnder, met: true };
   620→   620→      }
   621→   621→    }
   622→   622→  }
   623→   623→
   624→   624→  if (bestUnder) {
   625→   625→    report(0.92);
   626→   626→    return { best: bestUnder, met: true };
   627→   627→  }
   628→   628→
   629→   629→  // ─────────────────────────────────────────────────────────────────────
   630→   630→  // Phase 3 (enforceTarget / Max Compress only): guarantee the target is
   631→   631→  // met even at the cost of visual quality.
   632→   632→  //
   633→   633→  // Key improvements over the old aggressive search:
   634→   634→  //  • Quality ceiling is 0.6 (not 0.5) — recovers more fidelity when the
   635→   635→  //    scale is already small enough to satisfy the target.
   636→   636→  //  • Each scale refines upward in quality instead of accepting the
   637→   637→  //    binary-search midpoint, preventing unnecessary quality collapse.
   638→   638→  //  • Stops as soon as a near-ideal (within proximity band) result is
   639→   639→  //    found, so we don't keep shrinking past the point of usefulness.
   640→   640→  // ─────────────────────────────────────────────────────────────────────
   641→   641→  if (enforceTarget) {
   642→   642→    const aggressiveScales = slow
   643→   643→      ? [0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05]
   644→   644→      : [0.55, 0.5, 0.45, 0.4, 0.35, 0.3, 0.25, 0.2, 0.15, 0.1, 0.08, 0.06, 0.05, 0.04];
   645→   645→
   646→   646→    for (let si = 0; si < aggressiveScales.length; si++) {
   647→   647→      const scale = aggressiveScales[si];
   648→   648→      const frac = 0.92 + 0.06 * ((si + 1) / aggressiveScales.length);
   649→   649→
   650→   650→      if (skipQualitySearch) {
   651→   651→        const cand = await encode(source, origW, origH, scale, format, 1);
   652→   652→        report(frac);
   653→   653→        if (!cand) continue;
   654→   654→        if (cand.blob.size <= target) {
   655→   655→          bestUnder = pickBetterUnder(bestUnder, cand, target);
   656→   656→          if (isNearIdeal(bestUnder)) break;
   657→   657→        } else {
   658→   658→          bestAbove = pickSmallerAbove(bestAbove, cand);
   659→   659→        }
   660→   660→      } else {
   661→   661→        // Wider quality range than before [floor, 0.6] with refinement.
   662→   662→        let lo = AGGRESSIVE_QUALITY_FLOOR;
   663→   663→        let hi = 0.6;
   664→   664→        const aggressiveSteps = slow ? 4 : 7;
   665→   665→        for (let i = 0; i < aggressiveSteps; i++) {
   666→   666→          const mid = (lo + hi) / 2;
   667→   667→          const cand = await encode(source, origW, origH, scale, format, mid);
   668→   668→          report(frac);
   669→   669→          if (!cand) continue;
   670→   670→          if (cand.blob.size <= target) {
   671→   671→            bestUnder = pickBetterUnder(bestUnder, cand, target);
   672→   672→            lo = mid;
   673→   673→          } else {
   674→   674→            bestAbove = pickSmallerAbove(bestAbove, cand);
   675→   675→            hi = mid;
   676→   676→          }
   677→   677→          await yieldToMain();
   678→   678→        }
   679→   679→
   680→   680→        // Refine upward at this aggressive scale to recover quality.
   681→   681→        if (bestUnder && bestUnder.scale === scale) {
   682→   682→          bestUnder = await refineAroundBest(
   683→   683→            source, origW, origH, scale, format, bestUnder, bestUnder.quality, 0.6, target, report, frac, frac + 0.005,
   684→   684→          );
   685→   685→        }
   686→   686→
   687→   687→        if (isNearIdeal(bestUnder)) break;
   688→   688→      }
   689→   689→    }
   690→   690→
   691→   691→    // Last resort: extremely tiny scale.
   692→   692→    const lastResort = await encode(
   693→   693→      source, origW, origH, 0.02, format, skipQualitySearch ? 1 : AGGRESSIVE_QUALITY_FLOOR,
   694→   694→    );
   695→   695→    if (lastResort) {
   696→   696→      if (lastResort.blob.size <= target) {
   697→   697→        bestUnder = pickBetterUnder(bestUnder, lastResort, target);
   698→   698→      } else {
   699→   699→        bestAbove = pickSmallerAbove(bestAbove, lastResort);
   700→   700→      }
   701→   701→    }
   702→   702→
   703→   703→    if (bestUnder) {
   704→   704→      report(0.99);
   705→   705→      const q = Math.round(bestUnder.quality * 100);
   706→   706→      const s = Math.round(bestUnder.scale * 100);
   707→   707→      const ratio = bestUnder.blob.size / target;
   708→   708→      const note =
   709→   709→        ratio >= TARGET_PROXIMITY
   710→   710→          ? `Target ${formatBytes(target)} reached precisely (quality ${q}%, scale ${s}%).`
   711→   711→          : `Target ${formatBytes(target)} reached with aggressive compression (quality ${q}%, scale ${s}%). Image quality is significantly reduced.`;
   712→   712→      return { best: bestUnder, met: true, note };
   713→   713→    }
   714→   714→  }
   715→   715→
   716→   716→  // Truly impossible.
   717→   717→  report(1);
   718→   718→  const note =
   719→   719→    bestAbove && bestAbove.blob.size > target
   720→   720→      ? `Couldn't reach ${formatBytes(target)}. Closest result is ${formatBytes(bestAbove.blob.size)}.`
   721→   721→      : `Target size could not be reached. Consider a larger target or smaller image.`;
   722→   722→  return { best: bestAbove ?? bestUnder, met: false, note };
   723→   723→}
   724→   724→
   725→   725→/** Fine-grained upward refinement: given a candidate that is already under
   726→   726→ *  the target at (scale, quality), probe qualities just above `quality` to
   727→   727→ *  recover as much visual fidelity as possible without exceeding the target.
   728→   728→ *
   729→   729→ *  This is the key routine that prevents over-compression: instead of
   730→   730→ *  accepting the binary-search midpoint (which can be well below target),
   731→   731→ *  we push quality up until we're as close to the target as possible. */
   732→   732→async function refineAroundBest(
   733→   733→  source: CanvasImageSource,
   734→   734→  origW: number,
   735→   735→  origH: number,
   736→   736→  scale: number,
   737→   737→  format: string,
   738→   738→  current: EncodeCandidate,
   739→   739→  qLo: number,
   740→   740→  qHi: number,
   741→   741→  target: number,
   742→   742→  onProgress?: (p: number) => void,
   743→   743→  fracStart = 0.55,
   744→   744→  fracEnd = 0.62,
   745→   745→): Promise<EncodeCandidate> {
   746→   746→  let best = current;
   747→   747→  // Small number of fine steps — quality-to-size is monotonic, so a short
   748→   748→  // upward scan is enough to find the sweet spot.
   749→   749→  const fineSteps = 4;
   750→   750→  let lo = qLo;
   751→   751→  let hi = qHi;
   752→   752→  for (let i = 0; i < fineSteps; i++) {
   753→   753→    const mid = (lo + hi) / 2;
   754→   754→    // Skip if mid is at or below current best quality (no gain possible).
   755→   755→    if (mid <= best.quality + 0.005) break;
   756→   756→    const cand = await encode(source, origW, origH, scale, format, mid);
   757→   757→    onProgress?.(fracStart + (fracEnd - fracStart) * ((i + 1) / fineSteps));
   758→   758→    if (!cand) continue;
   759→   759→    if (cand.blob.size <= target) {
   760→   760→      // Still under target at higher quality → adopt and keep pushing up.
   761→   761→      best = cand;
   762→   762→      lo = mid;
   763→   763→    } else {
   764→   764→      // Over target at this quality → pull the ceiling down.
   765→   765→      hi = mid;
   766→   766→    }
   767→   767→    await yieldToMain();
   768→   768→  }
   769→   769→  return best;
   770→   770→}
   771→   771→
   772→   772→/** Pick the better of two candidates that are both under the target.
   773→   773→ *  Prefers the one closest to the target (largest size under target) so we
   774→   774→ *  never settle for a much smaller file when a nearer-to-target one exists.
   775→   775→ *  Ties in size favour higher quality. */
   776→   776→function pickBetterUnder(
   777→   777→  prev: EncodeCandidate | null,
   778→   778→  next: EncodeCandidate,
   779→   779→  target: number,
   780→   780→): EncodeCandidate {
   781→   781→  if (!prev) return next;
   782→   782→  // Both under target. Prefer the one closer to target = larger size.
   783→   783→  // Only prefer `next` if it is both larger (closer to target) AND not
   784→   784→  // exceeding the target. Quality breaks size ties.
   785→   785→  if (next.blob.size > prev.blob.size) return next;
   786→   786→  if (next.blob.size === prev.blob.size && next.quality > prev.quality) return next;
   787→   787→  return prev;
   788→   788→}
   789→   789→
   790→   790→function pickSmallerAbove(
   791→   791→  prev: EncodeCandidate | null,
   792→   792→  next: EncodeCandidate,
   793→   793→): EncodeCandidate {
   794→   794→  if (!prev) return next;
   795→   795→  return next.blob.size < prev.blob.size ? next : prev;
   796→   796→}

   798→   798→// Tiny local formatter to avoid a circular import with format.ts at runtime
   799→   799→// (kept inline so this module stays self-contained).
   800→   800→/** Yield to the event loop so the browser can process UI updates. */
   801→   801→function yieldToMain(): Promise<void> {
   802→   802→  return new Promise((r) => setTimeout(r, 0));
   803→   803→}

   805→   805→function formatBytes(bytes: number): string {
   806→   806→  if (!Number.isFinite(bytes) || bytes <= 0) return "0 B";
   807→   807→  const k = 1024;
   808→   808→  const units = ["B", "KB", "MB", "GB"];
   809→   809→  const i = Math.min(units.length - 1, Math.floor(Math.log(bytes) / Math.log(k)));
   810→   810→  const value = bytes / Math.pow(k, i);
   811→   811→  return `${value.toFixed(i === 0 ? 0 : 1)} ${units[i]}`;
   812→   812→}

   814→   814→function shortFormat(mime: string): string {
   815→   815→  switch (mime) {
   816→   816→    case "image/jpeg": return "JPG";
   817→   817→    case "image/png": return "PNG";
   818→   818→    case "image/webp": return "WebP";
   819→   819→    case "image/avif": return "AVIF";
   820→   820→    default: return mime.replace("image/", "").toUpperCase();
   821→   821→  }
   822→   822→}
